Formato do pacote

O campo subject deve usar um dos IDs reconhecidos pelo sistema. correct é o índice do gabarito: 0=A, 1=B, 2=C, 3=D.

    {
      "id": "oab48-etica-001",
      "subject": "etica",
      "topic": "Honorários",
      "text": "Enunciado...",
      "options": ["A", "B", "C", "D"],
      "correct": 1,
      "explanation": "Fundamento...",
      "source": "FGV/OAB",
      "exam": "48",
      "year": 2027,
      "official": true
    }
  ]
}
